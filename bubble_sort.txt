import java.util.*;
class Q10 {
    public static void main(String[] args) {
        Scanner sc=new Scanner(System.in);
        int a[]=new int[5];
        for(int i=0;i<5;i++) a[i]=sc.nextInt();
        System.out.println("Original: "+Arrays.toString(a));
        for(int i=0;i<5;i++)
            for(int j=0;j<4-i;j++)
                if(a[j]>a[j+1]) { int t=a[j]; a[j]=a[j+1]; a[j+1]=t; }
        System.out.println("Sorted: "+Arrays.toString(a));
        int key=sc.nextInt(), pos=-1, l=0, r=4;
        for(int i=0;i<5;i++) if(a[i]==key) pos=i;
        System.out.println("Linear Search: "+pos);
        while(l<=r) { int m=(l+r)/2; if(a[m]==key) { pos=m; break; } else if(a[m]<key) l=m+1; else r=m-1; }
        System.out.println("Binary Search: "+(l<=r?pos:-1));
    }
}
